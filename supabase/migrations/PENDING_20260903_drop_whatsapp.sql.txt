-- ---------------------------------------------------------------------
-- 0. PRE-FLIGHT — read-only, run these first and read the output
-- ---------------------------------------------------------------------
-- Confirms nothing new started depending on WhatsApp since this was drafted.

-- Expect 0 rows. Any row here is a tenant that would lose its lockbox
-- channel; the app now falls back to email, but fix the data anyway.
SELECT id, company_name, slug, lockbox_notification_methods
FROM tenants
WHERE lockbox_notification_methods::text ILIKE '%whatsapp%';

-- Expect 0. Meta WhatsApp was never configured by any tenant.
SELECT count(*) AS meta_configured
FROM tenants
WHERE integration_whatsapp
   OR meta_whatsapp_access_token IS NOT NULL
   OR meta_whatsapp_phone_number IS NOT NULL
   OR meta_whatsapp_phone_number_id IS NOT NULL
   OR meta_whatsapp_waba_id IS NOT NULL;

-- Snapshot before you drop anything. At drafting: 2 tenants
-- (East Peak Rentals LLC on the Twilio sandbox number, and the internal
-- "test" tenant).
SELECT company_name, slug, integration_twilio_whatsapp, twilio_whatsapp_number
FROM tenants
WHERE integration_twilio_whatsapp OR twilio_whatsapp_number IS NOT NULL;


-- ---------------------------------------------------------------------
-- 1. DATA — orphaned template rows on a channel nothing dispatches
-- ---------------------------------------------------------------------
-- 36 rows at drafting. These are unreachable from the portal composer now
-- (the channel tab is gone), and send-lead-message fails them loudly rather
-- than silently dropping them. Deleting is tidy-up, not a fix.
DELETE FROM lead_message_templates WHERE channel = 'whatsapp';

-- 1 row at drafting, belonging to the internal "test" tenant.
DELETE FROM lockbox_templates WHERE channel = 'whatsapp';

-- Normalise any tenant still saved as WhatsApp-only for lockbox delivery.
-- 1 tenant at drafting ("test"). The app already falls back to email, so
-- this only makes the stored value match the behaviour.
UPDATE tenants
SET lockbox_notification_methods = '["email"]'::jsonb
WHERE lockbox_notification_methods::text ILIKE '%whatsapp%';

-- NOT DELETED ON PURPOSE:
--   chat_channel_messages WHERE channel = 'whatsapp'  (26 rows, "test"
--   tenant) — this is delivered message history. The portal renders such a
--   row with default styling and no channel badge (channelMap lookup returns
--   undefined and the component returns null), so it degrades cleanly.
--   Deleting a customer's conversation history to tidy an enum is not worth it.

DROP TABLE IF EXISTS public.whatsapp_content_templates;


-- ---------------------------------------------------------------------
-- 3. COLUMNS — tenants
-- ---------------------------------------------------------------------
-- Meta / Cloud API side. NULL for all 57 tenants; integration_whatsapp
-- true for zero. This path never carried a message in production.
ALTER TABLE public.tenants DROP COLUMN IF EXISTS meta_whatsapp_access_token;
ALTER TABLE public.tenants DROP COLUMN IF EXISTS meta_whatsapp_phone_number;
ALTER TABLE public.tenants DROP COLUMN IF EXISTS meta_whatsapp_phone_number_id;
ALTER TABLE public.tenants DROP COLUMN IF EXISTS meta_whatsapp_waba_id;
ALTER TABLE public.tenants DROP COLUMN IF EXISTS integration_whatsapp;

-- Twilio side — this was the live one.
ALTER TABLE public.tenants DROP COLUMN IF EXISTS integration_twilio_whatsapp;
ALTER TABLE public.tenants DROP COLUMN IF EXISTS twilio_whatsapp_number;
ALTER TABLE public.tenants DROP COLUMN IF EXISTS twilio_whatsapp_lockbox_template_sid;


-- ---------------------------------------------------------------------
-- 4. COLUMNS — customers
-- ---------------------------------------------------------------------
-- Consent flag for a channel that no longer exists. 1 of 520 customers
-- opted in. Read by nothing after the removal.
ALTER TABLE public.customers DROP COLUMN IF EXISTS whatsapp_opt_in;
