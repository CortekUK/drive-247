-- Rollback for 20261019160100_multichat_sec_twilio_secrets_vault.sql
-- Run the rollback of 20261019160200_* FIRST (it restores the plaintext
-- columns from Vault); otherwise this drops the only copy of the secrets.

BEGIN;

DO $$
BEGIN
  IF EXISTS (SELECT 1 FROM public.tenants
              WHERE (twilio_auth_token_vault_id IS NOT NULL AND twilio_auth_token IS NULL)
                 OR (twilio_api_key_secret_vault_id IS NOT NULL AND twilio_api_key_secret IS NULL)) THEN
    RAISE EXCEPTION 'Plaintext columns are empty — roll back 20261019160200 first.';
  END IF;
END $$;

DELETE FROM vault.secrets
 WHERE id IN (SELECT twilio_auth_token_vault_id FROM public.tenants WHERE twilio_auth_token_vault_id IS NOT NULL
              UNION SELECT twilio_api_key_secret_vault_id FROM public.tenants WHERE twilio_api_key_secret_vault_id IS NOT NULL);

DROP FUNCTION IF EXISTS public.twilio_get_secrets(uuid);
DROP FUNCTION IF EXISTS public.twilio_set_secret(uuid, text, text);
ALTER TABLE public.tenants
  DROP COLUMN IF EXISTS twilio_auth_token_vault_id,
  DROP COLUMN IF EXISTS twilio_api_key_secret_vault_id;

COMMIT;
